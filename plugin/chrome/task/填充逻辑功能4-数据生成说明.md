### 2. 数据验证
添加数据验证逻辑，确保生成的 JSON 符合填充逻辑要求：

```javascript
function validateResumeData(data) {
  // 验证结构
  assert(Array.isArray(data), 'data 必须是数组');

  // 验证每个分组
  data.forEach(group => {
    assert(group.name, '分组必须有 name');
    assert(Array.isArray(group.fields), 'fields 必须是数组');

    // 验证每个字段
    group.fields.forEach(field => {
      assert(field.name, '字段必须有 name');
      assert(Array.isArray(field.blanks), 'blanks 必须是数组');
    });
  });
}
```
