## 🔧 插件代码示例

### 1. 打开登录窗口 (background.js)

```javascript
// 配置
const config = {
  WEB_LOGIN_URL: 'https://abc123.ngrok-free.app/login?from=plugin',
  API_BASE_URL: 'https://abc123.ngrok-free.app/api/plugin/',
  API_AUTOFILL_URL: 'https://abc123.ngrok-free.app/api/autofill/'
};

// 打开登录窗口
function openLoginWindow() {
  const loginWindow = window.open(
    config.WEB_LOGIN_URL,
    'plugin_login',
    'width=500,height=700,toolbar=no,menubar=no'
  );
  
  // 监听来自Web的消息
  window.addEventListener('message', async (event) => {
    if (event.data.type === 'PLUGIN_LOGIN_SUCCESS') {
      const { loginToken, userInfo } = event.data;
      
      try {
        // 用loginToken换取正式token
        const response = await fetch(`${config.API_BASE_URL}exchange-token`, {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ loginToken })
        });
        
        const data = await response.json();
        
        if (data.success) {
          // 存储token
          await chrome.storage.local.set({
            auth: {
              token: data.data.accessToken,
              refreshToken: data.data.refreshToken,
              userInfo: data.data.userInfo
            }
          });
          
          console.log('插件登录成功！', data.data.userInfo);
          
          // 通知popup登录成功
          chrome.runtime.sendMessage({ type: 'AUTH_SUCCESS' });
          
          loginWindow.close();
        } else {
          console.error('Token交换失败:', data.message);
        }
      } catch (error) {
        console.error('Token交换失败:', error);
      }
    }
  });
}
```

### 2. Token自动刷新 (background.js)

```javascript
// Token自动刷新
async function refreshTokenIfNeeded() {
  const { auth } = await chrome.storage.local.get('auth');
  
  if (!auth || !auth.refreshToken) {
    return false;
  }
  
  try {
    const response = await fetch(`${config.API_BASE_URL}refresh-token`, {
      method: 'POST',
      headers: {
        'Authorization': `Bearer ${auth.refreshToken}`
      }
    });
    
    const data = await response.json();
    
    if (data.success) {
      // 更新accessToken
      auth.token = data.data.accessToken;
      await chrome.storage.local.set({ auth });
      console.log('Token刷新成功');
      return true;
    }
  } catch (error) {
    console.error('Token刷新失败:', error);
  }
  
  return false;
}

// 请求拦截器：自动处理token过期
async function fetchWithAuth(url, options = {}) {
  const { auth } = await chrome.storage.local.get('auth');
  
  if (!auth || !auth.token) {
    throw new Error('未登录');
  }
  
  // 添加Authorization header
  options.headers = {
    ...options.headers,
    'Authorization': `Bearer ${auth.token}`
  };
  
  let response = await fetch(url, options);
  
  // 如果token过期（401），尝试刷新
  if (response.status === 401) {
    const refreshed = await refreshTokenIfNeeded();
    
    if (refreshed) {
      // 重新获取token
      const { auth: newAuth } = await chrome.storage.local.get('auth');
      options.headers['Authorization'] = `Bearer ${newAuth.token}`;
      
      // 重试请求
      response = await fetch(url, options);
    } else {
      // 刷新失败，需要重新登录
      throw new Error('登录已过期，请重新登录');
    }
  }
  
  return response;
}
```

### 3. 调用自动填充API (content.js)

```javascript
// 1. 分析页面
async function analyzePage() {
  const html = document.documentElement.outerHTML;
  const url = window.location.href;
  
  try {
    const response = await fetchWithAuth(
      `${config.API_AUTOFILL_URL}analyze-page`,
      {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ html, url })
      }
    );
    
    const data = await response.json();
    
    if (data.success) {
      return data.data; // { sessionId, fields }
    }
  } catch (error) {
    console.error('分析页面失败:', error);
  }
}

// 2. 获取简历列表
async function getResumeList() {
  try {
    const response = await fetchWithAuth(
      `${config.API_AUTOFILL_URL}resume/list`,
      { method: 'GET' }
    );
    
    const data = await response.json();
    
    if (data.success) {
      return data.data; // 简历列表
    }
  } catch (error) {
    console.error('获取简历列表失败:', error);
  }
}

// 3. 获取填充值
async function getFillValues(sessionId, fields, resumeData) {
  try {
    const response = await fetchWithAuth(
      `${config.API_AUTOFILL_URL}fill-values`,
      {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ sessionId, fields, resumeData })
      }
    );
    
    const data = await response.json();
    
    if (data.success) {
      return data.data; // { values }
    }
  } catch (error) {
    console.error('获取填充值失败:', error);
  }
}

// 4. 查询配额
async function getQuota() {
  try {
    const response = await fetchWithAuth(
      `${config.API_BASE_URL}get-quota`,
      {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({})
      }
    );
    
    const data = await response.json();
    
    if (data.success) {
      return data.data.remainQuota; // 剩余配额
    }
  } catch (error) {
    console.error('查询配额失败:', error);
  }
}
```

## 🔐 安全说明

### Token机制

1. **短期loginToken**：
   - 有效期：5分钟
   - 用途：仅用于插件与Web之间的一次性交换
   - 使用后立即失效

2. **accessToken**：
   - 有效期：1天
   - 用途：日常API调用
   - 过期后自动刷新

3. **refreshToken**：
   - 有效期：30天
   - 用途：刷新accessToken
   - 安全存储在 `chrome.storage.local`

### PostMessage安全

虽然示例代码中使用了 `'*'` 作为targetOrigin，但在生产环境中应该指定具体的域名：

```javascript
// ❌ 不安全（开发环境可以）
window.opener.postMessage(data, '*');

// ✅ 安全（生产环境）
window.opener.postMessage(data, 'chrome-extension://你的插件ID');
```

## 📝 注意事项

### 1. CORS配置

如果遇到跨域问题，确保后端配置了正确的CORS：

```java
@CrossOrigin(
    origins = {"https://abc123.ngrok-free.app", "chrome-extension://*"},
    allowedHeaders = "*",
    methods = {RequestMethod.GET, RequestMethod.POST, RequestMethod.PUT, RequestMethod.DELETE}
)
```

---

### 2. 内网穿透稳定性

- **Ngrok免费版**：每次重启域名会变化，需要重新配置
- **Ngrok付费版**：可以获得固定域名
- **Natapp免费版**：每次重启域名会变化
- **Natapp VIP**：可以获得固定域名

建议：
- 开发阶段使用免费版
- 生产环境建议购买付费版或部署到公网服务器

---

### 3. 插件manifest配置

确保插件的 `manifest.json` 包含必要的权限：

```json
{
  "manifest_version": 3,
  "permissions": [
    "storage",
    "tabs"
  ],
  "host_permissions": [
    "https://abc123.ngrok-free.app/*",
    "https://*/*",
    "http://*/*"
  ]
}
```

## 🐛 常见问题

### Q1: 登录窗口打开后无法通信？

**解决方案**：
- 检查 `window.opener` 是否存在
- 确保使用 `window.open()` 而不是 `chrome.windows.create()`
- 检查PostMessage的targetOrigin

---

### Q2: Token一直提示过期？

**解决方案**：
- 检查系统时间是否正确
- 查看后端日志确认token生成是否正常
- 确认token存储和读取逻辑正确

---

### Q3: API调用返回404？

**解决方案**：
- 检查API地址是否正确
- 确认后端是否启动
- 查看后端日志确认路由是否注册

---

### Q4: 内网穿透地址无法访问？

**解决方案**：
- 确认内网穿透工具正在运行
- 检查本地端口8080是否被占用
- 尝试重启内网穿透工具
