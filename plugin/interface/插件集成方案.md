# 🔌 Chrome插件集成方案

## 📋 目标
让内网另一台电脑的Chrome插件能够：
1. 通过内网穿透访问你的后端（localhost:8080）
2. 跳转到你的Web端进行登录认证
3. 调用自动填充API完成表单填充

---

## 🌐 第一步：配置内网穿透

### 方案A：使用 Ngrok（推荐）

#### 1. 下载并安装
```bash
# Windows
# 访问 https://ngrok.com/download
# 下载 Windows 版本并解压
```

#### 2. 注册并获取authtoken
```bash
# 访问 https://dashboard.ngrok.com/get-started/setup
# 注册账号，复制你的 authtoken
```

#### 3. 配置并启动
```bash
# 配置authtoken（只需一次）
ngrok authtoken 你的authtoken

# 启动内网穿透
ngrok http 8080
```

#### 4. 记录公网地址
启动后会显示：
```
Forwarding  https://1a2b-3c4d-5e6f.ngrok-free.app -> http://localhost:8080
```

## 🚀 第五步：插件登录流程实现

### 方案：OAuth式登录流程

#### 后端：新增插件登录回调接口

```java
@RestController
@RequestMapping("/api/plugin")
public class PluginAuthController {
    
    /**
     * 插件登录页面 - 跳转到Web登录
     */
    @GetMapping("/login")
    public String redirectToWebLogin() {
        // 重定向到前端登录页，带上特殊参数标识这是插件登录
        return "redirect:http://localhost:3000/login?from=plugin";
    }
    
    /**
     * 插件Token获取接口
     */
    @PostMapping("/get-token")
    public ResponseEntity<Map<String, Object>> getPluginToken(
        @RequestBody Map<String, Object> request
    ) {
        String loginToken = (String) request.get("loginToken");
        
        // 验证loginToken（短期一次性token）
        // 生成插件专用的accessToken和refreshToken
        // 返回token和用户信息
        
        Map<String, Object> response = new HashMap<>();
        response.put("accessToken", accessToken);
        response.put("refreshToken", refreshToken);
        response.put("userInfo", userInfo);
        
        return ResponseEntity.ok(response);
    }
}
```

#### 前端：Login.vue修改

```vue
<script setup>
// 检测是否是插件登录
const isPluginLogin = ref(false)

  const urlParams = new URLSearchParams(window.location.search)
  isPluginLogin.value = urlParams.get('from') === 'plugin'

  if (isPluginLogin.value) {

    // 通过PostMessage发送给插件
    if (window.opener) {

  }
```

```javascript
// 打开登录窗口
function openLoginWindow() {
  const loginUrl = 'https://your-domain.ngrok-free.app/login?from=plugin';
  const loginWindow = window.open(loginUrl, 'login', 'width=500,height=600');
  
  // 监听登录成功消息
  window.addEventListener('message', async (event) => {
    if (event.data.type === 'LOGIN_SUCCESS') {
      const { loginToken, userInfo } = event.data;
      
      // 使用loginToken换取正式token
      const response = await fetch('https://your-domain.ngrok-free.app/api/plugin/get-token', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ loginToken })
      });
      
      const data = await response.json();
      
      // 存储token
      chrome.storage.local.set({
        auth: {
          token: data.accessToken,
          refreshToken: data.refreshToken,
          userInfo: data.userInfo
        }
      });
      
      console.log('登录成功！');
      loginWindow.close();
    }
  });
}
```

## 🔑 插件登录流程代码

### 1. 后端新增接口

```java
@RestController
@RequestMapping("/api/plugin")
@CrossOrigin(origins = "*")
public class PluginAuthController {
    
    @Autowired
    private JwtUtil jwtUtil;
    
    /**
     * 生成插件登录token（一次性，5分钟有效）
     */
    @PostMapping("/generate-login-token")
    public ResponseEntity<Map<String, Object>> generateLoginToken(
        @RequestAttribute("userId") Long userId
    ) {
        // 生成短期token（5分钟）
        String loginToken = jwtUtil.generateShortToken(userId, 300);
        
        Map<String, Object> response = new HashMap<>();
        response.put("success", true);
        response.put("loginToken", loginToken);
        
        return ResponseEntity.ok(response);
    }
    
    /**
     * 用loginToken换取正式token
     */
    @PostMapping("/exchange-token")
    public ResponseEntity<Map<String, Object>> exchangeToken(
        @RequestBody Map<String, Object> request
    ) {
        String loginToken = (String) request.get("loginToken");
        
        // 验证loginToken
        Long userId = jwtUtil.validateToken(loginToken);
        
        if (userId == null) {
            return ResponseEntity.status(401).body(
                Map.of("success", false, "message", "登录token无效或已过期")
            );
        }
        
        // 生成正式的accessToken和refreshToken
        String accessToken = jwtUtil.generateToken(userId, null, null);  // 1天
        String refreshToken = jwtUtil.generateRefreshToken(userId);       // 30天
        
        // 获取用户信息
        User user = userService.getUserById(userId);
        
        Map<String, Object> response = new HashMap<>();
        response.put("success", true);
        response.put("accessToken", accessToken);
        response.put("refreshToken", refreshToken);
        response.put("userInfo", Map.of(
            "id", user.getId(),
            "nickname", user.getNickname(),
            "avatar", user.getAvatar()
        ));
        
        return ResponseEntity.ok(response);
    }
    
    /**
     * 刷新token
     */
    @PostMapping("/refresh-token")
    public ResponseEntity<Map<String, Object>> refreshToken(
        @RequestHeader("Authorization") String authHeader
    ) {
        String refreshToken = authHeader.replace("Bearer ", "");
        
        // 验证refreshToken
        Long userId = jwtUtil.validateRefreshToken(refreshToken);
        
        if (userId == null) {
            return ResponseEntity.status(401).body(
                Map.of("success", false, "message", "refresh token无效")
            );
        }
        
        // 生成新的accessToken
        String newAccessToken = jwtUtil.generateToken(userId, null, null);
        
        Map<String, Object> response = new HashMap<>();
        response.put("success", true);
        response.put("accessToken", newAccessToken);
        
        return ResponseEntity.ok(response);
    }
}
```

### 2. 前端Login.vue修改

```vue
<script setup>
import { ref, onMounted } from 'vue'
import { ElMessage } from 'element-plus'
import axios from 'axios'

const isPluginLogin = ref(false)

onMounted(() => {
  // 检测是否来自插件
  const urlParams = new URLSearchParams(window.location.search)
  isPluginLogin.value = urlParams.get('from') === 'plugin'
  
  if (isPluginLogin.value) {
    ElMessage.info('插件登录模式：登录后将自动返回')
  }
})

// 登录成功回调
const onLoginSuccess = async (token, userInfo) => {
  if (isPluginLogin.value) {
    try {
      // 生成一次性loginToken
      const response = await axios.post('/api/plugin/generate-login-token', {}, {
        headers: { 'Authorization': `Bearer ${token}` }
      })
      
      const loginToken = response.data.loginToken
      
      // 发送给插件
      if (window.opener) {
        window.opener.postMessage({
          type: 'PLUGIN_LOGIN_SUCCESS',
          loginToken: loginToken,
          userInfo: userInfo
        }, '*')
        
        ElMessage.success('登录成功！正在返回插件...')
        
        setTimeout(() => {
          window.close()
        }, 2000)
      }
    } catch (error) {
      ElMessage.error('生成登录token失败')
    }
  } else {
    // 正常Web登录
    localStorage.setItem('token', token)
    localStorage.setItem('userInfo', JSON.stringify(userInfo))
    router.push('/profile')
  }
}
</script>
```

```javascript
// 配置
const config = {
  API_BASE_URL: 'https://abc123.ngrok-free.app',
  WEB_LOGIN_URL: 'https://abc123.ngrok-free.app/login?from=plugin'
};

// 打开登录窗口
function openLoginWindow() {
  const loginWindow = window.open(
    config.WEB_LOGIN_URL,
    'plugin_login',
    'width=500,height=700,toolbar=no,menubar=no'
  );
  
  // 监听来自Web的消息
  window.addEventListener('message', async (event) => {
    // 验证来源（生产环境要严格验证）
    if (event.data.type === 'PLUGIN_LOGIN_SUCCESS') {
      const { loginToken, userInfo } = event.data;
      
      try {
        // 用loginToken换取正式token
        const response = await fetch(`${config.API_BASE_URL}/api/plugin/exchange-token`, {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ loginToken })
        });
        
        const data = await response.json();
        
        if (data.success) {
          // 存储token
          await chrome.storage.local.set({
            auth: {
              token: data.accessToken,
              refreshToken: data.refreshToken,
              userInfo: data.userInfo
            }
          });
          
          console.log('插件登录成功！', data.userInfo);
          
          // 通知popup登录成功
          chrome.runtime.sendMessage({ type: 'AUTH_SUCCESS' });
        }
      } catch (error) {
        console.error('Token交换失败:', error);
      }
    }
  });
}

// Token自动刷新
async function refreshTokenIfNeeded() {
  const { auth } = await chrome.storage.local.get('auth');
  
  if (!auth || !auth.refreshToken) {
    return false;
  }
  
  try {
    const response = await fetch(`${config.API_BASE_URL}/api/plugin/refresh-token`, {
      method: 'POST',
      headers: {
        'Authorization': `Bearer ${auth.refreshToken}`
      }
    });
    
    const data = await response.json();
    
    if (data.success) {
      // 更新accessToken
      auth.token = data.accessToken;
      await chrome.storage.local.set({ auth });
      return true;
    }
  } catch (error) {
    console.error('Token刷新失败:', error);
  }
  
  return false;
}
```
